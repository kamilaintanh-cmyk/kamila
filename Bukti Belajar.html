/* ===================================== BIODATA ONLINE - BUKTI BELAJAR KUMPULAN KALKULATOR MATEMATIKA ===================================== */ // ===================================== // 1. KONVERSI SISTEM BILANGAN // ===================================== function konversikan() { const angka = document.getElementById("angka").value.trim(); const asal = Number(document.getElementById("asal").value); const tujuan = Number(document.getElementById("tujuan").value); if (angka === "") { alert("Silakan masukkan angka!"); return; } const regex = { 2: /^[01]+$/, 8: /^[0-7]+$/, 10: /^\d+$/, 16: /^[0-9a-fA-F]+$/ }; if (!regex[asal].test(angka)) { alert("Angka tidak sesuai dengan sistem bilangan asal!"); return; } const nilai = parseInt(angka, asal); const hasil = nilai.toString(tujuan).toUpperCase(); document.getElementById("hasil").innerHTML = hasil; } // ===================================== // 2. ARITMETIKA MODULAR // ===================================== function hitungModular() { const a = Number(document.getElementById("nilaiA").value); const b = Number(document.getElementById("nilaiB").value); const m = Number(document.getElementById("modulus").value); const operasi = document.getElementById("operasi").value; if (!Number.isSafeInteger(a) || !Number.isSafeInteger(b) || !Number.isSafeInteger(m) || m <= 0) { alert("Masukkan bilangan bulat yang valid dan modulus lebih dari nol!"); return; } let hasil; switch (operasi) { case "tambah": hasil = a + b; break; case "kurang": hasil = a - b; break; case "kali": hasil = a * b; break; default: alert("Operasi tidak valid!"); return; } const hasilMod = ((hasil % m) + m) % m; document.getElementById("hasilModular").innerHTML = hasil + " mod " + m + " = " + hasilMod; } // ===================================== // 3. PENGECEK KONGRUENSI // ===================================== function cekKongruensi() { const a = Number(document.getElementById("kongruenA").value); const b = Number(document.getElementById("kongruenB").value); const m = Number(document.getElementById("kongruenMod").value); if (!Number.isSafeInteger(a) || !Number.isSafeInteger(b) || !Number.isSafeInteger(m) || m <= 0) { alert("Masukkan bilangan bulat yang valid!"); return; } const sisaA = ((a % m) + m) % m; const sisaB = ((b % m) + m) % m; let hasil; if (sisaA === sisaB) { hasil = `${a} ≡ ${b} (mod ${m}) - KONGRUEN`; } else { hasil = `${a} ≢ ${b} (mod ${m}) - TIDAK KONGRUEN`; } document.getElementById("hasilKongruensi").innerHTML = hasil; } // ===================================== // 4. FPB DAN KPK // ===================================== function hitungFPB(a, b) { a = Math.abs(a); b = Math.abs(b); while (b !== 0) { let sisa = a % b; a = b; b = sisa; } return a; } function hitungKPK() { const a = Number(document.getElementById("bilangan1").value); const b = Number(document.getElementById("bilangan2").value); if (!Number.isSafeInteger(a) || !Number.isSafeInteger(b) || a === 0 || b === 0) { alert("Masukkan dua bilangan bulat bukan nol!"); return; } const fpb = hitungFPB(a, b); const kpk = Math.abs((a / fpb) * b); document.getElementById("hasilFPB").innerHTML = "FPB = " + fpb + " | KPK = " + kpk; } // ===================================== // 5. PENGECEK BILANGAN PRIMA // ===================================== function cekPrima() { const angka = Number( document.getElementById("bilanganPrima").value ); if (!Number.isSafeInteger(angka)) { alert("Masukkan bilangan bulat yang valid!"); return; } let prima = angka >= 2; for (let i = 2; i <= Math.sqrt(angka) && prima; i++) { if (angka % i === 0) { prima = false; } } const hasil = prima ? angka + " adalah bilangan prima" : angka + " bukan bilangan prima"; document.getElementById("hasilPrima").innerHTML = hasil; } // ===================================== // 6. RESET SEMUA INPUT // ===================================== function resetKalkulator() { document.querySelectorAll("input").forEach(input => { input.value = ""; }); document.querySelectorAll(".hasil").forEach(hasil => { hasil.innerHTML = "-"; }); }
